let player; function loadYouTubeAPI() { const tag = document.createElement("script"); tag.src = "https://www.youtube.com/iframe_api"; document.head.appendChild(tag); } window.onYouTubeIframeAPIReady = function () { const iframe = document.querySelector(".youtube-wrap iframe"); if (!iframe) return; let src = iframe.getAttribute("src"); if (!src.includes("enablejsapi=1")) { src += "&enablejsapi=1"; iframe.setAttribute("src", src); } player = new YT.Player(iframe, { events: { onReady: function () { const playBtn = document.getElementById("playBtn"); if (playBtn) { playBtn.addEventListener("click", function () { player.playVideo(); }); } } } }); }; loadYouTubeAPI();
